N, C, G, H = map(int, input().split())
ranges = [tuple(map(int, input().split())) for _ in range(N)]

res = 0
for temp in range(-1, 1002):
    work = 0

    for i in range(len(ranges)):
        l, r = ranges[i]

        if temp < l:
            work += C
        if l <= temp <= r:
            work += G
        if temp > r:
            work += H

    res = max(res, work)

print(res)
