N = int(input())
a, b, c = map(int, input().split())

res = 0

for i in range(1, N + 1):
    for j in range(1, N + 1):
        for k in range(1, N + 1):
            # a와 i를 비교
            if abs(a - i) <= 2:
                res += 1

            # b와 j를 비교
            elif abs(b - j) <= 2:
                res += 1

            # c와 k를 비교
            elif abs(c - k) <= 2:
                res += 1

print(res)

2.
N = 6
abilities = list(map(int, input().split()))

res = 10000000

for i in range(0, N):
    for j in range(i + 1, N):
        for k in range(j + 1, N):
            # (i, j, k)가 A팀
            teamA = abilities[i] + abilities[j] + abilities[k]

            # 나머지가 B팀
            teamB = sum(abilities) - teamA

            res = min(res, abs(teamA - teamB))

print(res)

N = int(input())
arr_1 = list(map(int, input().split()))
arr_2 = list(map(int, input().split()))

cnt = 0
for i in range(1, N + 1):
    for j in range(1, N + 1):
        for k in range(1, N + 1):
            if (
                abs(arr_1[0] - i) <= 2 and abs(arr_1[1] - j) <= 2 and abs(arr_1[2] - k) <= 2
            ) or (
                abs(arr_2[0] - i) <= 2 and abs(arr_2[1] - j) <= 2 and abs(arr_2[2] - k) <= 2
            ):
                cnt += 1

print(cnt)
