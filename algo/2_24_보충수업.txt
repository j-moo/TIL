x = [p[0] for p in points]
y = [p[1] for p in points]

res = float('inf')

# 패스할 체크포인트 선택 => 양 끝의 2개는 선택하지 않는다.
for i in range(1, n - 1):

# 해당 체크포인트 외의 거리 계산
dist = 0

for j in range(1, n):
# 건너뛰는 경우
if j == i:
continue

# 전전 위치에서 이동하는 경우
if j == i + 1:
dist = dist + abs(x[j] - x[j - 2]) + abs(y[j] - y[j - 2])

# 전 위치에서 이동하는 경우
else:
dist = dist + abs(x[j] - x[j - 1]) + abs(y[j] - y[j - 1])

res = min(res, dist)

print(res)
